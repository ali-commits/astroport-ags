+// Catalog keys, units and types match the adapter's hello catalog.
+enum class Metric : uint8_t { GameFps, FrameTimeMs, Count };
+constexpr size_t MetricCount = static_cast<size_t>(Metric::Count);
+
+struct MetricInfo {
+	std::string_view key, unit, type;
+};
+inline constexpr std::array<MetricInfo, MetricCount> MetricCatalog {{
+    {"game_fps", "fps", "number"},
+    {"frame_time_ms", "ms", "number"},
+}};
+
+// The hello catalog says 1000: the fps window is aggregated once per second.
+constexpr uint32_t MetricsMinIntervalMs = 1000;
+constexpr uint32_t MetricsMaxIntervalMs = 3600000;
+
+using MetricMask = uint32_t; // Bit n is Metric n.
+constexpr MetricMask MetricBit(Metric metric) {
+	return MetricMask {1} << static_cast<unsigned>(metric);
+}
+
+struct MetricsSubscription {
+	MetricMask keys        = 0; // Empty means send nothing.
+	uint32_t   interval_ms = MetricsMinIntervalMs;
+};
+
+// Parses {"type":"metrics.subscribe","keys":[...],"interval_ms":n}. Unknown keys are ignored, an
+// empty list stops metrics, and the interval is clamped to the catalog minimum (a missing or
+// non-integer interval counts as the minimum). Returns false, leaving `output` alone, when `keys`
+// is absent or not an array: the host's packet is malformed and the old subscription stays.
+inline bool ParseMetricsSubscribe(const Json& packet, MetricsSubscription& output) noexcept {
+	if (!packet.is_object() || !packet.contains("keys") || !packet["keys"].is_array()) return false;
+	MetricsSubscription parsed;
+	for (const auto& key: packet["keys"]) {
+		if (!key.is_string()) continue;
+		const auto& name = key.get_ref<const std::string&>();
+		for (size_t i = 0; i < MetricCount; i++)
+			if (MetricCatalog[i].key == name) parsed.keys |= MetricMask {1} << i;
+	}
+	if (packet.contains("interval_ms") && packet["interval_ms"].is_number_integer()) {
+		const auto& interval = packet["interval_ms"];
+		if (interval.is_number_unsigned()) {
+			parsed.interval_ms = static_cast<uint32_t>(
+			    std::clamp<uint64_t>(interval.get<uint64_t>(), MetricsMinIntervalMs, MetricsMaxIntervalMs));
+		} else {
+			parsed.interval_ms = static_cast<uint32_t>(
+			    std::clamp<int64_t>(interval.get<int64_t>(), MetricsMinIntervalMs, MetricsMaxIntervalMs));
+		}
+	}
+	output = parsed;
+	return true;
+}
+
+struct MetricValues {
+	std::array<std::optional<double>, MetricCount> value {};
+	void Set(Metric metric, double v) { value[static_cast<size_t>(metric)] = v; }
+};
+
+// {"type":"metrics","t_ms":..,"values":{..}} with only subscribed keys that have a value.
+// Returns nullopt when nothing would be reported.
+inline std::optional<Json> BuildMetricsMessage(MetricMask subscribed, const MetricValues& values,
+                                               uint64_t t_ms) {
+	Json out = Json::object();
+	for (size_t i = 0; i < MetricCount; i++) {
+		if ((subscribed & (MetricMask {1} << i)) == 0 || !values.value[i]) continue;
+		out[std::string(MetricCatalog[i].key)] = *values.value[i];
+	}
+	if (out.empty()) return std::nullopt;
+	return Json {{"type", "metrics"}, {"t_ms", t_ms}, {"values", std::move(out)}};
+}
+
+// CLOCK_MONOTONIC in milliseconds.
+inline uint64_t MonotonicMs() noexcept {
+#if defined(CLOCK_MONOTONIC)
+	timespec now {};
+	clock_gettime(CLOCK_MONOTONIC, &now);
+	return static_cast<uint64_t>(now.tv_sec) * 1000 + static_cast<uint64_t>(now.tv_nsec) / 1000000;
+#else
+	return std::chrono::duration_cast<std::chrono::milliseconds>(
+	           std::chrono::steady_clock::now().time_since_epoch())
+	    .count();
+#endif
+}

+// When the next metrics message is due. The interval restarts after every attempt, whether or not
+// the send went through: under socket backpressure the source is read once per interval, not on
+// every pass of the 2 ms run loop.
+class MetricsPacer {
+public:
+	using Clock = std::chrono::steady_clock;
+	[[nodiscard]] bool Due(Clock::time_point now, uint32_t interval_ms) const {
+		return !m_last || now - *m_last >= std::chrono::milliseconds(interval_ms);
+	}
+	void Attempted(Clock::time_point now) { m_last = now; }
+	void Reset() { m_last.reset(); }
+
+private:
+	std::optional<Clock::time_point> m_last;
+};

+	std::function<MetricValues(MetricMask)> sample;
+};

+std::optional<Lease> LeaseRing::TryAcquire() {
+	std::unique_lock lock(m_mutex, std::try_to_lock);
+	if (!lock.owns_lock()) return {};
+	if (!m_connected || m_sequence == UINT64_MAX) return {};
+	for (uint32_t i = 0; i < SlotCount; i++)
+		if (m_leased[i] == 0) {
+			m_leased[i] = ++m_sequence;
+			Lease lease {m_epoch, m_sequence, i, m_used[i]};
+			m_used[i] = true;
+			return lease;
+		}
+	return {};
+}
+bool LeaseRing::Release(uint64_t epoch, uint32_t slot, uint64_t sequence) {
+	std::lock_guard lock(m_mutex);
+	if (!m_connected || epoch != m_epoch || slot >= SlotCount || sequence == 0 ||
+	    m_leased[slot] != sequence)
+		return false;
+	m_leased[slot] = 0;
+	return true;
+}
+void LeaseRing::Disconnect() {
+	std::lock_guard lock(m_mutex);
+	m_connected = false;
+}

+ReceiveResult ReceivePacket(int socket, Packet& packet, size_t expected_fds) {
+	char                  bytes[PacketLimit];
+	alignas(cmsghdr) char ancillary[CMSG_SPACE(sizeof(int) * 16)] {};
+	iovec                 iov {bytes, sizeof(bytes)};
+	msghdr                message {};

+static int Connect(const std::string& path) {

+	    status.st_uid != geteuid() || (status.st_mode & 077) != 0)

+	m_fds             = fds;

+void Transport::Publish(PendingFrame frame) {

+		return;

+	auto& pending = m_pending[frame.lease.slot];
+	pending.frame = frame;
+	pending.published.store(true, std::memory_order_release);

+	std::array<int, 9> fds;
+	fds.fill(fd);
+	const auto ring = std::make_shared<LeaseRing>(descriptor.epoch);
+	transport.SetPool(descriptor, fds, ring, generation);
+
+	Packet hello;
+	host.Receive(hello, 9);
+	PoolDescriptor parsed;
+	Check(ValidateHello(hello.json, hello.fds.size(), parsed), "valid v2 pool hello");
+	Check(hello.json["frames"] == true && parsed.epoch == 3, "pool hello carries frames and epoch");
+	host.Send({{"type", "refused"}, {"version", 2}, {"reason", "local"}});
+	host.ExpectFramelessHello();
+	host.AcceptFrameless(transport);
+	Check(!ring->TryAcquire(), "refused pool leases are retired");
+	close(fd);
+}

+void TestMetrics() {
+	MetricsSubscription subscription;
+	const Json first = {{"type", "metrics.subscribe"}, {"keys", {"game_fps", "nope"}},
+	                    {"interval_ms", 10}};
+	const Json second = {{"type", "metrics.subscribe"}, {"keys", {"frame_time_ms"}},
+	                     {"interval_ms", 1000}};
+	Check(ParseMetricsSubscribe(first, subscription), "parse metrics subscription");
+	Check(subscription.interval_ms == 1000, "metrics interval clamps to 1000");
+	MetricValues values;
+	values.Set(Metric::GameFps, 50);
+	values.Set(Metric::FrameTimeMs, 20);
+	const Json fps = {{"type", "metrics"}, {"t_ms", 123}, {"values", {{"game_fps", 50}}}};
+	Check(BuildMetricsMessage(subscription.keys, values, 123) == fps, "only game_fps subscribed");
+	Check(ParseMetricsSubscribe(second, subscription), "replace metrics subscription");
+	const Json frame_time = {{"type", "metrics"}, {"t_ms", 124}, {"values", {{"frame_time_ms", 20}}}};
+	Check(BuildMetricsMessage(subscription.keys, values, 124) == frame_time,
+	      "only frame_time_ms subscribed");
+
+	Host host;
+	Transport transport(host.path, {}, {{}, [values](MetricMask) { return values; }}, {}, true);
+	host.ConnectFrameless(transport);
+	host.Send(first);
+	{
+		Packet packet;
+		host.Receive(packet);
+		Check(packet.json["type"] == "metrics" && packet.json["values"] == fps["values"],
+		      "first subscription applies on the connection");
+	}
+	host.Send(second);
+	Packet packet;
+	host.Receive(packet);
+	Check(packet.json["type"] == "metrics" && packet.json["values"] == frame_time["values"],
+	      "second subscription replaces the first on the connection");
+}
