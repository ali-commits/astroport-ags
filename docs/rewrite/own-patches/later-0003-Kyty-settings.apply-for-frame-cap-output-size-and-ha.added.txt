+std::optional<std::pair<uint32_t, uint32_t>> ParseResolution(const Json& value) {
+	if (!value.is_string()) return std::nullopt;
+	const auto& text = value.get_ref<const std::string&>();
+	const auto separator = text.find('x');
+	if (separator == std::string::npos) return std::nullopt;
+
+	const auto parse_side = [](std::string_view side, uint32_t& size) {
+		const auto* end = side.data() + side.size();
+		const auto result = std::from_chars(side.data(), end, size);
+		return result.ec == std::errc{} && result.ptr == end && size >= 2 && size <= 8192;
+	};
+	uint32_t width = 0, height = 0;
+	const std::string_view resolution = text;
+	if (!parse_side(resolution.substr(0, separator), width) ||
+	    !parse_side(resolution.substr(separator + 1), height)) return std::nullopt;
+	return std::pair{width, height};
+}
+
+uint32_t ReadLaunchInteger(const char* key, uint32_t fallback, uint32_t min, uint32_t max) {
+	const char* value = std::getenv(key);
+	if (value == nullptr) return fallback;
+	char* end = nullptr;
+	const auto parsed = std::strtoul(value, &end, 10);
+	return end != value && *end == '\0' && parsed >= min && parsed <= max
+	           ? static_cast<uint32_t>(parsed) : fallback;
+}
+
+float LaunchHapticGain() {
+	const char* value = std::getenv("APS5_PAD_VIBRATION_GAIN");
+	if (value == nullptr) return 1.0f;
+	char* end = nullptr;
+	const float gain = std::strtof(value, &end);
+	return end != value && *end == '\0' && !std::isnan(gain) ? std::clamp(gain, 0.0f, 4.0f) : 1.0f;
+}

+Decision DecideSettingsApply(const Json& apply, const LiveState& now,
+                             std::span<const FrameCapRule> groups) {
+	Decision decision {{ {"type", "settings.applied"}, {"id", apply.value("id", Json{})},
+	                     {"applied", Json::object()}, {"values", Json::object()},
+	                     {"refused", Json::object()} }, {}, {}};
+	const auto requested = apply.find("values");
+	if (requested == apply.end() || !requested->is_object()) return decision;
+
+	auto& applied = decision.reply["applied"];
+	auto& refused = decision.reply["refused"];
+	for (const auto& [key, value]: requested->items()) {
+		if (key == "display.fps" && value.is_number_integer()) {
+			const auto fps = value.get<uint64_t>();
+			const auto rule = std::ranges::find(groups, fps, &FrameCapRule::fps);
+			if (rule != groups.end()) {
+				if (rule->fps != now.fps) {
+					const auto current = std::ranges::find(groups, now.fps, &FrameCapRule::fps);
+					if (current == groups.end() || current->group != rule->group) {
+						refused[key] = "Applies at the next start";
+						continue;
+					}
+					decision.cap_swap = std::pair{now.fps, rule->fps};
+				}
+				decision.actions.fps = rule->fps;
+				applied[key] = value;
+				continue;
+			}
+		} else if (key == "display.resolution") {
+			if (const auto size = ParseResolution(value)) {
+				if (now.fullscreen && !now.exporting) {
+					refused[key] = "Leave fullscreen first";
+				} else {
+					decision.actions.size = size;
+					applied[key] = value;
+				}
+				continue;
+			}
+		} else if (key == kHapticKey && value.is_number_integer()) {
+			const auto percent = value.get<uint64_t>();
+			if (percent >= 100 && percent <= 400) {
+				decision.actions.haptic = static_cast<float>(percent) / 100.0f;
+				decision.reply["values"][key] = value;
+				continue;
+			}
+		}
+		refused[key] = "Not supported";
+	}
+	return decision;
+}
+
+LiveStateStorage& Live() {

+// LiveState is too large for a lock-free atomic. Keep the whole snapshot under one lock.
+class LiveStateStorage {
+public:
+	explicit LiveStateStorage(LiveState state): m_state(state) {}
+
+	LiveState load() const {
+		std::lock_guard lock(m_mutex);
+		return m_state;
+	}

+	void store(LiveState state) {
+		Update([&](LiveState& current) { current = state; });
+	}
+
+	template <typename UpdateState>
+	void Update(UpdateState update) {
+		std::lock_guard lock(m_mutex);
+		update(m_state);
+	}

+int LivePatch(const char* name, bool on) {
+#if defined(__linux__)
+	using Patch = int32_t (*)(const char*, int32_t);
+	static const auto patch = []() -> Patch {
+		const char* audit = std::getenv("LD_AUDIT");
+		if (audit == nullptr) return nullptr;
+		std::string_view entries(audit);
+		while (!entries.empty()) {
+			const auto separator = entries.find_first_of(": ");
+			const auto path = entries.substr(0, separator);
+			if (path.ends_with("libastro_patcher.so")) {
+				void* library = dlopen(std::string(path).c_str(), RTLD_NOW | RTLD_LOCAL);
+				if (library == nullptr) return nullptr;
+				auto symbol = reinterpret_cast<Patch>(dlsym(library, "astro_patch_live"));
+				if (symbol == nullptr) dlclose(library);
+				return symbol;
+			}
+			entries.remove_prefix(separator == std::string_view::npos ? entries.size() : separator + 1);
+		}
+		return nullptr;
+	}();
+	if (patch != nullptr) return patch(name, on ? 1 : 0);
+#else
+	(void)name;
+	(void)on;
+#endif
+	return -10;
+}

+Json ApplySettings(const Json& apply) {
+	const auto now = Live().load();
+	auto decision = DecideSettingsApply(apply, now);
+	auto& actions = decision.actions;
+
+	if (decision.cap_swap) {
+		const auto [old_fps, new_fps] = *decision.cap_swap;
+		const auto old_cap = "precise frame cap " + std::to_string(old_fps) + " fps";
+		const auto new_cap = "precise frame cap " + std::to_string(new_fps) + " fps";
+		// shortcut: ASTROHYBRID_LOW_LATENCY_INTERVAL_US keeps its launch value; update it when a group gets a second value
+		if (LivePatch(old_cap.c_str(), false) != 0 || LivePatch(new_cap.c_str(), true) != 0) {
+			LivePatch(old_cap.c_str(), true);
+			decision.reply["applied"].erase("display.fps");
+			decision.reply["refused"]["display.fps"] = "Applies at the next start";
+			actions.fps.reset();
+		}
+	}
+
+	if (actions.fps) Libs::VideoOut::SetFrameLimit(*actions.fps);
+	if (actions.size && !now.exporting) {
+		const auto [width, height] = *actions.size;
+		Libs::Graphics::WindowRequestSize(width, height);
+	}
+	Live().Update([&](LiveState& state) {
+		if (actions.fps) state.fps = *actions.fps;
+		if (actions.size) {
+			state.width = actions.size->first;
+			state.height = actions.size->second;
+		}
+		if (actions.haptic) state.haptic = *actions.haptic;
+	});
+	return std::move(decision.reply);
+}
