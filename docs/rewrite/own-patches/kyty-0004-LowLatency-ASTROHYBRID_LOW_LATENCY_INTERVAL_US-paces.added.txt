+	// ASTROHYBRID_LOW_LATENCY_INTERVAL_US (default 0: no limit): the driver also paces frames to at least
+	// this interval in vkLatencySleepNV, before the game starts the next one (33333 = 30 FPS), so the
+	// frame cap sits where Reflex lowers latency instead of after the frame was simulated.
+	static const uint32_t interval = [] {
+		const char* value = std::getenv("ASTROHYBRID_LOW_LATENCY_INTERVAL_US");
+		const long  us    = value != nullptr ? std::strtol(value, nullptr, 10) : 0;
+		return us > 0 && us < 1000000 ? static_cast<uint32_t>(us) : 0u;
+	}();
+	mode.minimumIntervalUs = Mode() == 1 ? interval : 0;
